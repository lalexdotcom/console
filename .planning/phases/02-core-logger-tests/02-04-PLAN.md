import { describe, expect, test } from '@rstest/core';
import { L, LogLevels } from '../../../src';
import { captureAll } from '../../helpers/capture';

// reset.ts is registered globally via rstest.config.ts setupFiles.
// beforeEach: registry.scopes is cleared → each test using L.scope() gets a fresh scope.

describe('Scope creation (SCOPE-01)', () => {
  test('L.scope() returns a ScopeLogger with scope property and all level methods', () => {
    const s = L.scope('scope-01-api');
    // scope property
    expect(s.scope).toBe('scope-01-api');
    // all 11 level methods
    for (const level of LogLevels) {
      expect(typeof (s as unknown as Record<string, unknown>)[level]).toBe('function');
    }
    // mixin methods
    expect(typeof s.once).toBe('function');
    expect(typeof s.limit).toBe('function');
    expect(typeof s.options).toBe('function');
  });

  test('level methods on a scope actually emit output', () => {
    L.format = 'json';
    const s = L.scope('scope-01-emit');
    const { stdout } = captureAll(() => s.info('from scope'));
    expect(stdout).toHaveLength(1);
    const parsed = JSON.parse(stdout[0].trimEnd()) as Record<string, unknown>;
    expect(parsed.scope).toBe('scope-01-emit');
    expect(parsed.severity).toBe('info');
    expect(parsed.msg).toBe('from scope');
  });
});

describe('Scope caching (SCOPE-02)', () => {
  test('same scope name returns cached instance (strict identity equality)', () => {
    const s1 = L.scope('scope-cache-01');
    const s2 = L.scope('scope-cache-01');
    expect(s1).toBe(s2); // Object.is equality — same reference
  });

  test('different scope names return different instances', () => {
    const a = L.scope('scope-diff-a');
    const b = L.scope('scope-diff-b');
    expect(a).not.toBe(b);
    expect(a.scope).toBe('scope-diff-a');
    expect(b.scope).toBe('scope-diff-b');
  });
});

describe('Scope option inheritance (SCOPE-03)', () => {
  test('scope inherits root options when no own override is set', () => {
    L.date = true; // rootOptions.date = true
    const s = L.scope('scope-inherit-01');
    expect(s.date).toBe(true); // cascaded from rootOptions
  });

  test('scope own option overrides root option', () => {
    L.date = true; // root
    const s = L.scope('scope-inherit-02');
    s.date = false; // own override
    expect(s.date).toBe(false); // own wins
  });

  test('overriding scope option does not mutate root', () => {
    L.date = true;
    const s = L.scope('scope-inherit-03');
    s.date = false;
    expect(L.date).toBe(true);  // root unchanged
    expect(s.date).toBe(false); // scope override applied
  });

  test('scope inherits root level and receives level-filtered output', () => {
    L.format = 'json';
    L.level = 'warn'; // root: suppress info, debug, etc.
    const s = L.scope('scope-inherit-level');
    // No own level set → inherits root 'warn' filtering
    const { stdout } = captureAll(() => s.info('filtered'));
    expect(stdout).toHaveLength(0); // info(7) > warn(4) threshold → suppressed
  });
});

describe('Scope mutation isolation (SCOPE-04)', () => {
  test('mutating one scope does not affect a sibling scope', () => {
    const s1 = L.scope('scope-isolate-01');
    const s2 = L.scope('scope-isolate-02');
    s1.level = 'error';
    // s2 has no own level → cascades to root default (wth)
    expect(s2.level).toBe('wth'); // unaffected — default from root
  });

  test('mutating a scope does not affect root', () => {
    const s = L.scope('scope-isolate-root');
    s.level = 'error';
    s.date = true;
    expect(L.level).toBe('wth');  // root default unchanged
    expect(L.date).toBe(false);   // root default unchanged
  });

  test('output from s1 after s2 mutation still reflects s1 settings', () => {
    L.format = 'json';
    const s1 = L.scope('scope-isolate-out-01');
    const s2 = L.scope('scope-isolate-out-02');
    // Suppress s2 but not s1
    s2.enabled = false;
    const { stdout } = captureAll(() => {
      s1.info('should emit');
      s2.info('should NOT emit');
    });
    expect(stdout).toHaveLength(1);
    const parsed = JSON.parse(stdout[0].trimEnd()) as Record<string, unknown>;
    expect(parsed.scope).toBe('scope-isolate-out-01');
  });
});

```typescript
import { describe, expect, test } from '@rstest/core';
import { L } from '../../../src';
import { captureAll } from '../../helpers/capture';

// reset.ts is registered globally via rstest.config.ts setupFiles.
//
// CRITICAL: Use L.scope('unique-per-test-name') for every mixin test (per D-09).
// After reset.ts beforeEach, registry.scopes is cleared. Next L.scope('name') call
// creates a fresh scope with a fresh entries Map in the createLimitMixin closure.
// This guarantees counters start at 0 for each test regardless of test ordering.

describe('once() (MIX-01)', () => {
  test('once() emits exactly once per call-site regardless of repeat calls', () => {
    L.format = 'json';
    const s = L.scope('mix-once-01');
    const { stdout } = captureAll(() => {
      // All 5 calls are at the same source line → same call-site key → one counter
      for (let i = 0; i < 5; i++) s.once().info('repeated');
    });
    expect(stdout).toHaveLength(1);
  });

  test('once() uses separate counters for different call-sites', () => {
    L.format = 'json';
    const s = L.scope('mix-once-02');
    const { stdout } = captureAll(() => {
      s.once().info('site-A'); // different source line from site-B
      s.once().info('site-B'); // different source line from site-A
    });
    // Two different call-site lines → two independent once() counters → 2 emissions
    expect(stdout).toHaveLength(2);
  });
});

describe('limit(n) (MIX-02)', () => {
  test('limit(3) emits exactly 3 times per call-site across 10 calls', () => {
    L.format = 'json';
    const s = L.scope('mix-limit-01');
    const { stdout } = captureAll(() => {
      for (let i = 0; i < 10; i++) s.limit(3).info('limited');
      // Same source line → same call-site key → counter incremented per call
    });
    expect(stdout).toHaveLength(3);
  });

  test('limit(1) behaves like once()', () => {
    L.format = 'json';
    const s = L.scope('mix-limit-02');
    const { stdout } = captureAll(() => {
      for (let i = 0; i < 5; i++) s.limit(1).info('once-limit');
    });
    expect(stdout).toHaveLength(1);
  });

  test('limit(n) counter does not exceed n even with different message args', () => {
    L.format = 'json';
    const s = L.scope('mix-limit-03');
    const { stdout } = captureAll(() => {
      for (let i = 0; i < 10; i++) s.limit(2).info(`msg-${i}`);
    });
    expect(stdout).toHaveLength(2);
  });
});

describe('limit() with explicit key (MIX-03)', () => {
  // Explicit key bypasses getLimitCallerKey() — multiple call-sites share one counter.

  test('explicit key groups disparate call-sites under one counter', () => {
    L.format = 'json';
    const s = L.scope('mix-limit-key-01');
    const { stdout } = captureAll(() => {
      s.limit(2, 'shared').info('first call-site');   // different line from next
      s.limit(2, 'shared').info('second call-site');  // same key → shared counter=2
      s.limit(2, 'shared').info('third call-site');   // counter=2 reached → dropped
    });
    expect(stdout).toHaveLength(2); // 3rd call dropped: counter at max
  });

  test('different explicit keys are independent', () => {
    L.format = 'json';
    const s = L.scope('mix-limit-key-02');
    const { stdout } = captureAll(() => {
      s.limit(1, 'key-a').info('a1');
      s.limit(1, 'key-a').info('a2'); // dropped: key-a counter=1
      s.limit(1, 'key-b').info('b1');
      s.limit(1, 'key-b').info('b2'); // dropped: key-b counter=1
    });
    expect(stdout).toHaveLength(2); // a1 and b1 pass
  });
});

describe('options({}) one-shot override (MIX-04)', () => {
  // createOverrideMixin returns a GenericLogger that uses overrides as one-shot EmitOptions.
  // NOTE: 'level' is excluded from the overrides type (Partial<Omit<LoggerOptions, 'level'>>).
  // Use observable options like 'date' to verify one-shot behavior.

  test('options({date:true}) applies date bracket for one call then reverts', () => {
    L.format = 'pretty';
    L.pad = false;
    const s = L.scope('mix-opts-01');
    // s.date defaults to false (no own setting, root default=false)
    expect(s.date).toBe(false);

    // One-shot: date=true for this single call
    const { stdout: shot } = captureAll(() =>
      s.options({ date: true }).info('one-shot'),
    );
    expect(shot[0]).toMatch(/\[\d{4}-\d{2}-\d{2}/); // date bracket present

    // Next call: one-shot is gone, s.date still false
    const { stdout: next } = captureAll(() => s.info('reverted'));
    expect(next[0]).not.toMatch(/\[\d{4}-\d{2}-\d{2}/); // no date bracket
  });

  test('options({date:true}) does not mutate scope options', () => {
    const s = L.scope('mix-opts-02');
    s.options({ date: true }).info('one-shot');
    expect(s.date).toBe(false); // scope date still false — not mutated
  });

  test('options({color:false}) applies override for one call only', () => {
    L.format = 'json';
    const s = L.scope('mix-opts-03');
    // Verify options override reaches the emit path (no throw, output produced)
    const { stdout } = captureAll(() =>
      s.options({ color: false }).info('color-off'),
    );
    expect(stdout).toHaveLength(1);
    // color=false doesn't affect json output, but call must succeed
    const parsed = JSON.parse(stdout[0].trimEnd()) as Record<string, unknown>;
    expect(parsed.msg).toBe('color-off');
    expect(s.color).toBe(true); // scope color not mutated
  });
});
```
