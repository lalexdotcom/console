```typescript
import { afterEach, describe, expect, test } from '@rstest/core';
import { L } from '../../../src';
import { captureAll } from '../../helpers/capture';

// reset.ts is registered globally via rstest.config.ts setupFiles.
//
// CRITICAL: reset.ts does NOT reset console patch state or bypass state.
// Every test in this file that calls L.patch() MUST call L.unpatch() in cleanup.
// Every test that calls L.bypass() MUST call L.restore() in cleanup.
// Use try/finally in each test body for reliable cleanup.

// Global safety net: unpatch and restore after every test in this file,
// even if a test assertion throws before reaching the cleanup.
afterEach(() => {
  L.unpatch();
  L.restore();
});

describe('patch() (CONS-01)', () => {
  test('patch() replaces console.log with logger dispatch', () => {
    L.format = 'json';
    L.patch();
    try {
      const { stdout } = captureAll(() => console.log('patched via console'));
      expect(stdout).toHaveLength(1);
      const parsed = JSON.parse(stdout[0].trimEnd()) as Record<string, unknown>;
      // console.log → L.info → severity='info'
      expect(parsed.severity).toBe('info');
      expect(parsed.msg).toBe('patched via console');
    } finally {
      L.unpatch();
    }
  });

  test('patch() replaces console.error with logger dispatch to stderr', () => {
    L.format = 'json';
    L.patch();
    try {
      const { stderr } = captureAll(() => console.error('error via console'));
      expect(stderr).toHaveLength(1);
      const parsed = JSON.parse(stderr[0].trimEnd()) as Record<string, unknown>;
      expect(parsed.severity).toBe('error');
      expect(parsed.msg).toBe('error via console');
    } finally {
      L.unpatch();
    }
  });

  test('patch() replaces console.warn with logger dispatch to stderr', () => {
    L.format = 'json';
    L.patch();
    try {
      const { stderr } = captureAll(() => console.warn('warning via console'));
      expect(stderr).toHaveLength(1);
      const parsed = JSON.parse(stderr[0].trimEnd()) as Record<string, unknown>;
      expect(parsed.severity).toBe('warn');
    } finally {
      L.unpatch();
    }
  });
});

describe('unpatch() (CONS-02)', () => {
  test('unpatch() restores original console.log behavior', () => {
    L.patch();
    L.unpatch();
    // After unpatch, console.log writes plain unstructured text — not JSON
    const { stdout } = captureAll(() => console.log('raw output'));
    expect(stdout).toHaveLength(1);
    // Plain console.log output is NOT parseable as JSON
    expect(() => JSON.parse(stdout[0].trimEnd())).toThrow();
  });

  test('unpatch() is idempotent — calling twice does not throw', () => {
    L.patch();
    L.unpatch();
    expect(() => L.unpatch()).not.toThrow();
  });
});

describe('bypass() (CONS-03)', () => {
  test('bypass(spy) redirects L.info output to spy.info — not to process.stdout', () => {
    L.format = 'json';
    const calls: Array<{ method: string; args: unknown[] }> = [];
    const spy = {
      log:   (...args: unknown[]) => calls.push({ method: 'log',   args }),
      info:  (...args: unknown[]) => calls.push({ method: 'info',  args }),
      debug: (...args: unknown[]) => calls.push({ method: 'debug', args }),
      warn:  (...args: unknown[]) => calls.push({ method: 'warn',  args }),
      error: (...args: unknown[]) => calls.push({ method: 'error', args }),
    } as unknown as Console;

    L.bypass(spy);
    try {
      // captureAll captures process.stdout/stderr — bypass routes AWAY from these
      const { stdout, stderr } = captureAll(() => L.info('bypassed'));
      // Nothing written to stdout/stderr — all went to spy
      expect(stdout).toHaveLength(0);
      expect(stderr).toHaveLength(0);
      // Spy received the call
      expect(calls).toHaveLength(1);
      expect(calls[0].method).toBe('info');
      // args[0] is the JSON line (json format, console method receives the serialized string)
      const parsed = JSON.parse(calls[0].args[0] as string) as Record<string, unknown>;
      expect(parsed.msg).toBe('bypassed');
      expect(parsed.severity).toBe('info');
    } finally {
      L.restore();
    }
  });

  test('bypass() routes error-level output to spy.error', () => {
    L.format = 'json';
    const calls: Array<{ method: string; args: unknown[] }> = [];
    const spy = {
      error: (...args: unknown[]) => calls.push({ method: 'error', args }),
    } as unknown as Console;

    L.bypass(spy);
    try {
      captureAll(() => L.error('err'));
      expect(calls).toHaveLength(1);
      expect(calls[0].method).toBe('error');
    } finally {
      L.restore();
    }
  });
});

describe('restore() (CONS-04)', () => {
  test('restore() reverts bypass — subsequent L.info calls go to process.stdout', () => {
    L.format = 'json';
    const spy = { info: () => {} } as unknown as Console;
    L.bypass(spy);
    L.restore();
    // After restore, output flows to system console → captured by captureAll
    const { stdout } = captureAll(() => L.info('restored'));
    expect(stdout).toHaveLength(1);
    const parsed = JSON.parse(stdout[0].trimEnd()) as Record<string, unknown>;
    expect(parsed.msg).toBe('restored');
  });

  test('restore() is idempotent — calling twice does not throw', () => {
    L.bypass({ info: () => {} } as unknown as Console);
    L.restore();
    expect(() => L.restore()).not.toThrow();
  });
});
```

Notes:
- `afterEach(() => { L.unpatch(); L.restore(); })` provides a safety net so that even if a test assertion throws before the `finally` cleanup, the global console and bypass state are always restored.
- `captureAll` captures `process.stdout.write` / `process.stderr.write`. After `L.bypass(spy)`, `method.apply(activeConsole, args)` routes through `spy` not `systemConsole` — so `captureAll` receives nothing (stdout/stderr = empty). This is the correct observable behavior for CONS-03.
- `rstest.config.ts` has `disableConsoleIntercept: true` — rstest itself does not interfere with console method patching.
