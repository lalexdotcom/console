**Step 3 — Create `tests/helpers/reset.ts` (singleton registry reset — per D-07):**
```typescript
import { beforeEach } from '@rstest/core';

const REGISTRY_KEY = '$logger-registry';

/**
 * Resets the logger singleton registry before each test.
 * Clears scopes, exclusive lock, format, and root option overrides in place.
 * The root logger object itself is preserved (same reference), but all
 * accumulated state is wiped so each test starts from default behavior.
 */
beforeEach(() => {
  const g = globalThis as Record<string, unknown>;
  const reg = g[REGISTRY_KEY] as
    | {
        scopes: Record<string, unknown>;
        exclusive?: unknown;
        format: string;
        rootOptions: Record<string, unknown>;
      }
    | undefined;

  if (reg) {
    for (const key of Object.keys(reg.scopes)) {
      delete reg.scopes[key];
    }
    delete reg.exclusive;
    reg.format = 'json';
    for (const key of Object.keys(reg.rootOptions)) {
      delete reg.rootOptions[key];
    }
  }
});
```

Design rationale (agent's discretion area from CONTEXT.md): The registry `const` inside `src/logger/index.ts` is captured by closure in the IIFE — deleting from `globalThis` won't affect the existing reference. Instead, we reset the object IN PLACE: clear scopes cache, remove exclusive lock, restore default format, and wipe rootOptions so `computeOptions` cascades to `DEFAULT_LOGGER_OPTIONS`. The root logger object is kept (same reference) because `createLogger()` is not exported. rstest's per-file module isolation (`isolate: true` default) provides file-level isolation; this `beforeEach` provides within-file test isolation.

**Create `tests/helpers/stdout.ts` (per D-03, D-04 — one file, functional wrapper):**
```typescript
/**
 * Captures all process.stdout.write calls during callback execution.
 * Returns an array of written strings. Synchronous — the logger's dispatch is sync.
 *
 * @param fn - Callback to execute while stdout is intercepted.
 * @returns Array of string chunks written to stdout.
 */
export function captureStdout(fn: () => void): string[] {
  const chunks: string[] = [];
  const original = process.stdout.write.bind(process.stdout);

  process.stdout.write = ((chunk: string | Uint8Array): boolean => {
    chunks.push(typeof chunk === 'string' ? chunk : new TextDecoder().decode(chunk));
    return true;
  }) as typeof process.stdout.write;

  try {
    fn();
  } finally {
    process.stdout.write = original;
  }

  return chunks;
}
```

Key design details:
- Uses `.bind(process.stdout)` to preserve `this` context when saving the original.
- Casts the replacement to `typeof process.stdout.write` because the actual signature has multiple overloads.
- Always restores original in `finally` — no shared mutable state (per D-04).
- Synchronous only — the logger dispatch path is synchronous (console.log/warn/error are sync).

**Create `tests/helpers/console-spy.ts` (per D-03, D-04, D-06):**
```typescript
/** A captured console method call with the method name and arguments. */
interface ConsoleCall {
  method: string;
  args: unknown[];
}

/** Console methods to spy on — covers all methods the logger dispatches to. */
const SPY_METHODS = ['log', 'info', 'debug', 'warn', 'error', 'groupCollapsed', 'groupEnd'] as const;

/**
 * Spies on console methods during callback execution.
 * Returns an array of { method, args } for each captured call.
 * Synchronous — the logger's browser dispatch is sync.
 *
 * @param fn - Callback to execute while console is intercepted.
 * @returns Array of ConsoleCall objects.
 */
export function spyOnConsole(fn: () => void): ConsoleCall[] {
  const calls: ConsoleCall[] = [];
  const originals = Object.fromEntries(
    SPY_METHODS.map((m) => [m, console[m]]),
  ) as Record<(typeof SPY_METHODS)[number], typeof console.log>;

  for (const method of SPY_METHODS) {
    (console as Record<string, unknown>)[method] = (...args: unknown[]) => {
      calls.push({ method, args });
    };
  }

  try {
    fn();
  } finally {
    for (const method of SPY_METHODS) {
      (console as Record<string, unknown>)[method] = originals[method];
    }
  }

  return calls;
}

export type { ConsoleCall };
```

Key design details:
- Spies on `groupCollapsed` and `groupEnd` in addition to log/info/debug/warn/error — the logger uses `groupCollapsed` for TRACE_LEVELS in browser mode.
- Each invocation returns a fresh array — no shared mutable state between calls (per D-04).
- Originals are restored in `finally` — safe even if the callback throws.
- `ConsoleCall` type is exported for use in test assertions.
