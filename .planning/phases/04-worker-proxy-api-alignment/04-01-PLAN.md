describe('unserializable args fallback (WORK-03)', () => {
  test('function arg is converted via String() — not structuredClone-able', async () => {
    // Functions throw DataCloneError in structuredClone; cloneArg falls back to String().
    // String(() => 42) returns the source representation (e.g. "() => 42").
    WL.info((() => 42) as unknown as string);
    await flush();

    const msgs = getLogMsgs();
    expect(msgs.length).toBeGreaterThanOrEqual(1);
    const arg = msgs[msgs.length - 1].args[0];
    // String() succeeded → result is a string, not '[unserializable]'
    expect(typeof arg).toBe('string');
    expect(arg as string).not.toBe('[unserializable]');
  });

  test('arg where both structuredClone AND String() throw → "[unserializable]"', async () => {
    // A function with overridden toString/toPrimitive that throws:
    // - structuredClone(evilFn) → DataCloneError (functions are not cloneable)
    // - String(evilFn) → calls evilFn[Symbol.toPrimitive] → throws → propagates
    // → cloneArg catch block returns '[unserializable]'
    const evilFn = Object.assign(
      // biome-ignore lint/suspicious/noEmptyBlockStatements: intentional stub function
      function () {},
      {
        toString() {
          throw new Error('cannot stringify');
        },
        [Symbol.toPrimitive]() {
          throw new Error('cannot stringify');
        },
      },
    );

    WL.info(evilFn as unknown as string);
    await flush();

    const msgs = getLogMsgs();
    expect(msgs.length).toBeGreaterThanOrEqual(1);
    expect(msgs[msgs.length - 1].args[0]).toBe('[unserializable]');
